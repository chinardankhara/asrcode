9. COMMON USAGE PATTERNS
================================================================================

9.1 BASIC AGENT INVOCATION
---------------------------
    import boto3
    client = boto3.client('bedrock-agentcore')

    response = client.invoke_agent_runtime(
        agentId='agent-id',
        sessionId='session-id',
        inputText='User query'
    )

9.2 MEMORY MANAGEMENT WORKFLOW
-------------------------------
    # Create memory
    client.batch_create_memory_records(
        memoryRecords=[
            {'content': 'Memory content', 'metadata': {...}}
        ]
    )

    # Retrieve memory
    response = client.retrieve_memory_records(
        query='search query'
    )

    # Update memory
    client.batch_update_memory_records(
        updates=[
            {'recordId': 'id', 'content': 'new content'}
        ]
    )

9.3 SESSION LIFECYCLE
---------------------
    # Start session
    session = client.start_code_interpreter_session()
    session_id = session['sessionId']

    # Use session
    result = client.invoke_code_interpreter(
        sessionId=session_id,
        code='print("Hello World")'
    )

    # Stop session
    client.stop_code_interpreter_session(
        sessionId=session_id
    )

9.4 EVENT TRACKING
------------------
    # Create event
    client.create_event(
        eventType='agent_action',
        payload={'action': 'task_completed'}
    )

    # List events
    paginator = client.get_paginator('list_events')
    for page in paginator.paginate():
        for event in page['events']:
            print(f"Event: {event['eventType']}")
